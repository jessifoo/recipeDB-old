# FPIES-Friendly Recipe Platform

A modern recipe platform designed for families managing FPIES and food allergies. Built with Next.js 14 and Flask.

## Tech Stack

### Frontend
- Next.js 14 with App Router
- TypeScript for type safety
- Tailwind CSS for styling
- Modern React patterns and hooks

### Backend
- Flask for the REST API
- SQLAlchemy for database operations
- PostgreSQL for data storage

## Features
- Search and filter recipes by allergens
- Track liked recipes
- Modern, responsive UI
- Type-safe development
- Fast, server-side rendered pages

## Recipe Management

### Viewing Recipes
The application provides a modern, filterable interface for viewing recipes at `/recipes`. Features include:

- **Recipe Cards**: Each recipe is displayed as a card showing:
  - Title and meal type
  - Cooking method and times
  - Preview image (when available)
  - Allergen information (dairy-free, soy-free, etc.)

- **Filtering Options**:
  - Search by recipe title
  - Filter by meal type (breakfast, lunch, dinner, etc.)
  - Filter by cooking method (stovetop, bake, grill, etc.)
  - Filter by allergens (dairy-free, soy-free, gluten-free, etc.)
  - View recipe collections separately

### Recent Updates
- Added recipe database with support for allergen tracking
- Implemented recipe import functionality
- Created filterable recipe list view with modern UI

import { Recipe, RecipeFilters } from '@/types/recipe';

export const filterRecipes = (recipes: Recipe[], filters: RecipeFilters): Recipe[] => {
  return recipes.filter(recipe => {
    // Filter by search query
    if (filters.searchQuery) {
      const query = filters.searchQuery.toLowerCase();
      const matchesTitle = recipe.title.toLowerCase().includes(query);
      const matchesDescription = recipe.description.toLowerCase().includes(query);
      if (!matchesTitle && !matchesDescription) return false;
    }

    // Filter by allergens
    if (filters.allergens?.length) {
      const hasAllergen = recipe.allergens.some(allergen =>
        filters.allergens?.includes(allergen)
      );
      if (hasAllergen) return false;
    }

    return true;
  });
};

export const paginateRecipes = (
  recipes: Recipe[],
  page: number = 1,
  perPage: number = 12
): Recipe[] => {
  const start = (page - 1) * perPage;
  const end = start + perPage;
  return recipes.slice(start, end);
};

## frontend/components/recipes/RecipeFilters.tsx
```
import { useState } from 'react';
import { Search, X } from 'lucide-react';
import { RecipeFilters } from '@/types/recipe';

const COMMON_ALLERGENS = [
  'Milk',
  'Eggs',
  'Fish',
  'Shellfish',
  'Tree Nuts',
  'Peanuts',
  'Wheat',
  'Soy',
  'Rice',
  'Oats',
];

interface FilterProps {
  onFilterChange: (filters: RecipeFilters) => void;
}

export function RecipeFilters({ onFilterChange }: FilterProps) {
  const [selectedAllergens, setSelectedAllergens] = useState<string[]>([]);
  const [searchQuery, setSearchQuery] = useState('');

  const handleAllergenToggle = (allergen: string) => {
    setSelectedAllergens((prev) => {
      const newAllergens = prev.includes(allergen)
        ? prev.filter((a) => a !== allergen)
        : [...prev, allergen];

      onFilterChange({
        allergens: newAllergens,
        searchQuery,
      });

      return newAllergens;
    });
  };

  const handleSearch = (query: string) => {
    setSearchQuery(query);
    onFilterChange({
      allergens: selectedAllergens,
      searchQuery: query,
    });
  };

  return (
    <div className="space-y-6">
      {/* Search Bar */}
      <div className="relative">
        <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 h-5 w-5" />
        <input
          type="text"
          placeholder="Search recipes..."
          value={searchQuery}
          onChange={(e) => handleSearch(e.target.value)}
          className="w-full pl-10 pr-4 py-2 border rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500 outline-none"
        />
      </div>

      {/* Allergen Filters */}
      <div>
        <h3 className="text-lg font-medium mb-3">Filter by Allergens</h3>
        <div className="flex flex-wrap gap-2">
          {COMMON_ALLERGENS.map((allergen) => {
            const isSelected = selectedAllergens.includes(allergen);
            return (
              <button
                key={allergen}
                onClick={() => handleAllergenToggle(allergen)}
                className={`px-3 py-1 rounded-full text-sm font-medium transition-colors ${
                  isSelected
                    ? 'bg-blue-100 text-blue-800'
                    : 'bg-gray-100 text-gray-700 hover:bg-gray-200'
                }`}
              >
                {allergen}
                {isSelected && (
                  <X className="inline-block ml-1 h-3 w-3" />
                )}
              </button>
            );
          })}
        </div>
      </div>
    </div>
  );
}

  isEggFree?: boolean;
  searchQuery?: string;
  isCollection?: boolean;
}

  const [filters, setFilters] = useState<RecipeFilters>({});

  useEffect(() => {
    fetchRecipes();
  }, [filters]);

  const fetchRecipes = async () => {
    try {
      const queryParams = new URLSearchParams();

      if (filters.mealType) {
        queryParams.append('meal_type', filters.mealType);
      }
      if (filters.cookingMethod) {
        queryParams.append('cooking_method', filters.cookingMethod);
      }
      if (filters.searchQuery) {
        queryParams.append('q', filters.searchQuery);
      }
      if (filters.isCollection !== undefined) {
        queryParams.append('is_collection', filters.isCollection.toString());
      }

      // Add allergen filters
      const allergens: string[] = [];
      if (filters.isDairyFree) allergens.push('dairy');
      if (filters.isSoyFree) allergens.push('soy');
      if (filters.isGlutenFree) allergens.push('gluten');
      if (filters.isNutFree) allergens.push('nuts');
      if (filters.isEggFree) allergens.push('eggs');

      allergens.forEach(allergen => {
        queryParams.append('allergens', allergen);
      });

      const response = await fetch(`/api/recipes?${queryParams.toString()}`);
      const data = await response.json();
      setRecipes(data);
    } catch (error) {
      console.error('Error fetching recipes:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleFilterChange = (field: keyof RecipeFilters, value: any) => {
    setFilters(prev => ({ ...prev, [field]: value }));
  };

  return (
    <Box sx={{ p: 3 }}>
      {/* Filters */}
      <Box sx={{ mb: 4 }}>
        <Grid container spacing={2} alignItems="center">
          <Grid item xs={12} sm={3}>
            <TextField
              fullWidth
              label="Search Recipes"
              value={filters.searchQuery || ''}
              onChange={(e) => handleFilterChange('searchQuery', e.target.value)}
            />
          </Grid>
          <Grid item xs={12} sm={3}>
            <FormControl fullWidth>
              <InputLabel>Meal Type</InputLabel>
              <Select
                value={filters.mealType || ''}
                label="Meal Type"
                onChange={(e) => handleFilterChange('mealType', e.target.value)}
              >
                <MenuItem value="">All</MenuItem>
                {mealTypes.map(type => (
                  <MenuItem key={type} value={type}>
                    {type.charAt(0).toUpperCase() + type.slice(1)}
                  </MenuItem>
                ))}
              </Select>
            </FormControl>
          </Grid>
          <Grid item xs={12} sm={3}>
            <FormControl fullWidth>
              <InputLabel>Cooking Method</InputLabel>
              <Select
                value={filters.cookingMethod || ''}
                label="Cooking Method"
                onChange={(e) => handleFilterChange('cookingMethod', e.target.value)}
              >
                <MenuItem value="">All</MenuItem>
                {cookingMethods.map(method => (
                  <MenuItem key={method} value={method}>
                    {method.split('_').map(word => word.charAt(0).toUpperCase() + word.slice(1)).join(' ')}
                  </MenuItem>
                ))}
              </Select>
            </FormControl>
          </Grid>
          <Grid item xs={12}>
            <Stack direction="row" spacing={2}>
              <FormControlLabel
                control={
                  <Checkbox
                    checked={filters.isDairyFree || false}
                    onChange={(e) => handleFilterChange('isDairyFree', e.target.checked)}
                  />
                }
                label="Dairy Free"
              />
              <FormControlLabel
                control={
                  <Checkbox
                    checked={filters.isSoyFree || false}
                    onChange={(e) => handleFilterChange('isSoyFree', e.target.checked)}
                  />
                }
                label="Soy Free"
              />
              <FormControlLabel
                control={
                  <Checkbox
                    checked={filters.isGlutenFree || false}
                    onChange={(e) => handleFilterChange('isGlutenFree', e.target.checked)}
                  />
                }
                label="Gluten Free"
              />

                }

                label="Collections Only"
              />

      </Box>

      <Grid container spacing={3}>
        {recipes.map((recipe) => (
          <Grid item key={recipe.id} xs={12} sm={6} md={4}>
            <Card sx={{ height: '100%', display: 'flex', flexDirection: 'column' }}>

                  image={recipe.image_preview_url}
                  alt={recipe.title}
                  sx={{ objectFit: 'cover' }}
                />
              )}
              <CardContent sx={{ flexGrow: 1 }}>
                <Typography gutterBottom variant="h6" component="h2">
                  {recipe.title}
                </Typography>
                <Stack direction="row" spacing={1} sx={{ mb: 1 }}>
                  {recipe.meal_type && (
                    <Chip

                      size="small"

                    />
                  )}
                  {recipe.cooking_method && (

              </CardContent>
            </Card>
          </Grid>
        ))}
      </Grid>

## frontend/app/page.tsx
```
'use client';

import { useState } from 'react';
import { RecipeCard } from '@/components/RecipeCard';
import { RecipeFilters } from '@/components/RecipeFilters';
import type { Recipe, RecipeFilters as Filters } from '@/types/recipe';

export default function Home() {
  const [recipes, setRecipes] = useState<Recipe[]>([]);
  const [loading, setLoading] = useState(false);

  const handleFilterChange = async (filters: Filters) => {
    setLoading(true);
    try {
      const params = new URLSearchParams();
      if (filters.searchQuery) params.append('q', filters.searchQuery);
      if (filters.allergens?.length) {
        filters.allergens.forEach(allergen =>
          params.append('allergens', allergen)
        );
      }

      const response = await fetch(`/api/recipes?${params.toString()}`);
      const data = await response.json();
      setRecipes(data);
    } catch (error) {
      console.error('Failed to fetch recipes:', error);
    } finally {
      setLoading(false);
    }
  };

  const handleLike = async (recipeId: number) => {
    try {
      await fetch(`/api/recipes/${recipeId}/like`, {
        method: 'POST',
      });

      setRecipes(prev =>
        prev.map(recipe =>
          recipe.id === recipeId
            ? { ...recipe, isLiked: !recipe.isLiked }
            : recipe
        )
      );
    } catch (error) {
      console.error('Failed to like recipe:', error);
    }
  };

  return (
    <main className="container mx-auto px-4 py-8">
      <h1 className="text-4xl font-bold text-gray-900 mb-8">
        FPIES-Friendly Recipes
      </h1>

      <div className="mb-8">
        <RecipeFilters onFilterChange={handleFilterChange} />
      </div>

      {loading ? (
        <div className="text-center py-12">
          <div className="animate-spin rounded-full h-12 w-12 border-b-2 border-gray-900 mx-auto"></div>
          <p className="mt-4 text-gray-600">Loading recipes...</p>
        </div>
      ) : recipes.length > 0 ? (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
          {recipes.map((recipe) => (
            <RecipeCard
              key={recipe.id}
              recipe={recipe}
              onLike={handleLike}
            />
          ))}
        </div>
      ) : (
        <div className="text-center py-12">
          <p className="text-gray-600">
            No recipes found. Try adjusting your filters.
          </p>
        </div>
      )}
    </main>
  );
}

import csv
import sys

# Add the parent directory to PYTHONPATH so we can import app

def normalize_cooking_method(method):
    """Normalize cooking method to match our enum"""
    if not method:
        return None

    method = method.lower()
    if 'bake' in method:
        return 'bake'
    elif 'stovetop' in method or 'simmer' in method or 'fry' in method or 'pan' in method:
        return 'stovetop'
    elif 'grill' in method:
        return 'grill'
    elif 'slow cooker' in method or 'crockpot' in method:
        return 'slow_cooker'
    elif 'instant pot' in method:
        return 'instant_pot'
    elif 'air fry' in method:
        return 'air_fryer'
    elif 'refrigerate' in method:
        return 'refrigerate'
    elif 'freeze' in method:
        return 'freeze'
    return 'stovetop'  # default

def detect_allergens_from_title(title):
    """Detect allergen information from recipe title"""
    title = title.lower()
    return {
        'is_dairy_free': 'dairy-free' in title or 'dairy free' in title,
        'is_soy_free': 'soy-free' in title or 'soy free' in title,
        'is_gluten_free': 'gluten-free' in title or 'gluten free' in title,
        'is_nut_free': 'nut-free' in title or 'nut free' in title,
        'is_egg_free': 'egg-free' in title or 'egg free' in title
    }

def import_recipes(csv_path):
    """Import recipes from CSV file"""
    app = create_app()

    with app.app_context():
        # First, clear existing recipes
        Recipe.query.delete()
        db.session.commit()

        # Read and import recipes
        with open(csv_path, 'r', encoding='utf-8-sig') as f:  # Use utf-8-sig to handle BOM
            reader = csv.DictReader(f)
            count = 0
            for row in reader:
                # Skip empty rows
                if not row['Recipe Title']:
                    continue

                # Convert times to integers
                prep_time = int(row['prep time']) if row['prep time'] and row['prep time'].isdigit() else None
                cook_time = int(row['cook time']) if row['cook time'] and row['cook time'].isdigit() else None

                # Clean up title (some have markdown-style numbering)
                title = row['Recipe Title']
                if title.startswith(('1.', '2.', '3.')):
                    title = title.split('**')[-1].strip(' :*')

                # Detect allergens from title
                allergen_info = detect_allergens_from_title(title)

                recipe = Recipe(
                    title=title,
                    meal_type=row['Meal Type'].lower() if row['Meal Type'] else 'other',
                    is_collection=row['Meal Type'].lower() == 'collection',
                    source_url=row['URL'],
                    image_url=row['Image'],
                    image_preview_url=row['Image Preview URL'],
                    ingredients=[],  # We'll need to scrape these from the source URL
                    instructions=[],  # We'll need to scrape these from the source URL
                    prep_time=prep_time,
                    cook_time=cook_time,
                    cooking_method=normalize_cooking_method(row['cook method']),
                    date_added=datetime.utcnow(),
                    **allergen_info
                )

                db.session.add(recipe)
                count += 1

                # Commit in batches to avoid memory issues
                if count % 10 == 0:
                    db.session.commit()
                    print(f"Imported {count} recipes...")

            # Final commit for remaining recipes
            db.session.commit()
            print(f"Successfully imported {count} recipes")

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print("Usage: python import_recipes.py <path_to_csv>")
        sys.exit(1)

    csv_path = sys.argv[1]
    if not os.path.exists(csv_path):
        print(f"Error: File {csv_path} does not exist")
        sys.exit(1)

    import_recipes(csv_path)

async def search_recipes(
    session: AsyncSession,
    query: str,
    filters: Dict[str, Any],
    max_age: Optional[timedelta] = None
) -> List[Recipe]:
    """
    Search recipes in database with filters

    Args:
        session: Database session
        query: Search query
        filters: Search filters
        max_age: Maximum age for cached recipes

    Returns:
        List of matching recipes

    Raises:
        DatabaseError: If database operation fails
    """
    try:
        async with session.begin_nested():
            stmt = select(Recipe)

            # Apply text search on title and ingredients
            if query:
                stmt = stmt.where(or_(
                    Recipe.title.ilike(f"%{query}%"),
                    Recipe.ingredients.cast(String).ilike(f"%{query}%")
                ))

            # Apply filters
            if filters:
                filter_conditions = []

                if filters.get('cuisine'):
                    filter_conditions.append(
                        Recipe.cuisines.any(Cuisine.name == filters['cuisine'])
                    )

                if filters.get('diet'):
                    filter_conditions.append(
                        Recipe.diets.any(Diet.name == filters['diet'])
                    )

                if filters.get('max_ready_time'):
                    filter_conditions.append(
                        Recipe.ready_in_minutes <= filters['max_ready_time']
                    )

                if filters.get('protein_type'):
                    filter_conditions.append(
                        Recipe.protein_type == filters['protein_type']
                    )

                if filters.get('cooking_method'):
                    filter_conditions.append(
                        Recipe.cooking_method == filters['cooking_method']
                    )

                if filters.get('allergens'):
                    for allergen in filters['allergens']:
                        filter_conditions.append(
                            ~Recipe.allergens.any(Allergen.name == allergen)
                        )

                if filter_conditions:
                    stmt = stmt.where(and_(*filter_conditions))

            # Apply cache age filter
            if max_age:
                min_fetch_time = datetime.utcnow() - max_age
                stmt = stmt.where(Recipe.last_fetched >= min_fetch_time)

            # Execute query
            result = await session.execute(stmt)
            recipes = result.scalars().all()

            logger.debug(
                "Recipe search completed",
                query=query,
                filters=filters,
                max_age=max_age,
                result_count=len(recipes)
            )
            return list(recipes)

    except SQLAlchemyError as e:
        logger.error(
            "Failed to search recipes",
            query=query,
            filters=filters,
            error=str(e),
            error_type=type(e).__name__,
            exc_info=True
        )
        raise DatabaseError(f"Failed to search recipes: {str(e)}", original_error=e)

from typing import Any, Dict, Optional
import asyncio
import aiohttp
from tenacity import retry, stop_after_attempt, wait_exponential
from app.config.api_config import APIConfig, APIEndpoint
from app.config.logging_config import get_logger
from .exceptions import APIError, ConfigurationError

logger = get_logger(__name__)

class APIClient:
    """Generic API client with retry and error handling"""

    def __init__(self, session: aiohttp.ClientSession, api_name: str):
        """
        Initialize API client

        Args:
            session: aiohttp client session
            api_name: Name of the API (e.g., 'spoonacular')
        """
        self.session = session
        self.api_name = api_name
        self.config = APIConfig()
        self.endpoint = self.config.get_endpoint(api_name)

        # Set up auth headers based on API requirements
        self.default_headers = self._get_auth_headers()

        logger.info(
            "Initialized API client",
            api_name=api_name,
            base_url=self.endpoint.base_url
        )

    def _get_auth_headers(self) -> Dict[str, str]:
        """Get authentication headers for the API"""
        headers = {}

        # Handle different API authentication methods
        if self.api_name == 'spoonacular':
            # Spoonacular uses query parameter for API key
            pass
        elif self.api_name == 'edamam':
            # Edamam uses query parameters for app_id and app_key
            pass
        elif self.api_name == 'api_ninjas':
            # API Ninjas uses X-Api-Key header
            api_key = self.config.get_api_key(self.api_name, 'API_NINJAS_API_KEY')
            headers['X-Api-Key'] = api_key
        elif self.api_name == 'tasty':
            # Tasty uses RapidAPI headers
            api_key = self.config.get_api_key(self.api_name, 'TASTY_API_KEY')
            headers['X-RapidAPI-Key'] = api_key
            headers['X-RapidAPI-Host'] = 'tasty.p.rapidapi.com'

        return headers

    def _get_auth_params(self) -> Dict[str, str]:
        """Get authentication parameters for APIs that use query parameters"""
        params = {}

        if self.api_name == 'spoonacular':
            api_key = self.config.get_api_key(self.api_name, 'SPOONACULAR_API_KEY')
            params['apiKey'] = api_key
        elif self.api_name == 'edamam':
            app_id = self.config.get_api_key(self.api_name, 'EDAMAM_APP_ID')
            app_key = self.config.get_api_key(self.api_name, 'EDAMAM_APP_KEY')
            params.update({
                'app_id': app_id,
                'app_key': app_key
            })

        return params

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=4, max=10),
        retry_error_callback=lambda retry_state: retry_state.outcome.result()
    )
    async def request(
        self,
        method: str,
        path: str,
        params: Optional[Dict[str, Any]] = None,
        headers: Optional[Dict[str, str]] = None,
        json: Optional[Dict[str, Any]] = None,
        timeout: Optional[int] = None
    ) -> Dict[str, Any]:
        """
        Make an API request with retry and error handling

        Args:
            method: HTTP method (GET, POST, etc.)
            path: URL path to append to base URL
            params: Query parameters
            headers: Request headers
            json: JSON body for POST/PUT requests
            timeout: Request timeout in seconds

        Returns:
            API response as dictionary

        Raises:
            APIError: For API-related errors
            ConfigurationError: For configuration issues
        """
        url = f"{self.endpoint.base_url}/{path.lstrip('/')}"
        timeout = timeout or self.endpoint.timeout

        # Merge authentication parameters
        request_params = self._get_auth_params()
        if params:
            request_params.update(params)

        # Merge headers with defaults
        request_headers = dict(self.default_headers)
        if headers:
            request_headers.update(headers)

        try:
            async with self.session.request(
                method=method,
                url=url,
                params=request_params,
                headers=request_headers,
                json=json,
                timeout=aiohttp.ClientTimeout(total=timeout)
            ) as response:
                # Log request details
                logger.info(
                    "API request",
                    method=method,
                    url=url,
                    status=response.status,
                    params={k: v for k, v in request_params.items() if not k.lower().endswith('key')}
                )

                # Handle different status codes
                if response.status == 200:
                    return await response.json()
                elif response.status == 429:
                    retry_after = response.headers.get('Retry-After', '60')
                    logger.warning(
                        "Rate limit exceeded",
                        api_name=self.api_name,
                        retry_after=retry_after
                    )
                    raise APIError(
                        "Rate limit exceeded",
                        self.api_name,
                        response.status
                    )
                elif response.status >= 400:
                    error_text = await response.text()
                    logger.error(
                        "API error response",
                        api_name=self.api_name,
                        status=response.status,
                        error=error_text
                    )
                    raise APIError(
                        f"API error: {error_text}",
                        self.api_name,
                        response.status
                    )

        except aiohttp.ClientError as e:
            logger.error(
                "API request failed",
                api_name=self.api_name,
                error=str(e),
                url=url,
                method=method
            )
            raise APIError(
                f"Request failed: {str(e)}",
                self.api_name
            ) from e
        except asyncio.TimeoutError as e:
            logger.error(
                "API request timeout",
                api_name=self.api_name,
                url=url,
                method=method,
                timeout=timeout
            )
            raise APIError(
                f"Request timed out after {timeout}s",
                self.api_name
            ) from e

        await self.cache.clear()

    def cached(self, ttl: Optional[int] = None) -> Callable:
        """
        Decorator for caching async function results

        Args:
            ttl: Optional time-to-live in seconds

        Returns:
            Decorated function that uses caching
        """
        def decorator(func: Callable[..., Awaitable[T]]) -> Callable[..., Awaitable[T]]:
            @wraps(func)
            async def wrapper(*args: Any, **kwargs: Any) -> T:
                # Create cache key from function name and arguments
                key = f"{func.__name__}:{str(args)}:{str(kwargs)}"

                # Try to get from cache
                cached_value = await self.get(key)
                if cached_value is not None:
                    return cached_value

                # If not in cache, call function and cache result
                result = await func(*args, **kwargs)
                await self.set(key, result, ttl)
                return result

            return wrapper
        return decorator

"""Rate limiting implementation for API requests"""
import time
from collections import deque
from dataclasses import dataclass, field
from typing import Deque, Dict
from .exceptions import RateLimitError

@dataclass
class RateLimitWindow:
    """Represents a rate limit window with a maximum number of requests"""
    max_requests: int
    window_seconds: int
    requests: Deque[float] = field(default_factory=deque)

    def add_request(self) -> None:
        """Add a request timestamp to the window"""
        now = time.time()
        self.cleanup_old_requests(now)
        self.requests.append(now)

    def cleanup_old_requests(self, current_time: float) -> None:
        """Remove requests older than the window"""
        while self.requests and current_time - self.requests[0] >= self.window_seconds:
            self.requests.popleft()

    def can_make_request(self) -> bool:
        """Check if a request can be made within the current window"""
        self.cleanup_old_requests(time.time())
        return len(self.requests) < self.max_requests

class APIRateLimiter:
    """Rate limiter for API requests with multiple time windows"""

    def __init__(self, api_name: str):
        """Initialize rate limiter for specific API"""
        self.api_name = api_name
        self.windows: Dict[str, RateLimitWindow] = {}

    def add_limit(self, name: str, max_requests: int, window_seconds: int) -> None:
        """Add a rate limit window"""
        self.windows[name] = RateLimitWindow(max_requests, window_seconds)

    def check_rate_limit(self) -> None:
        """
        Check if the request can be made within all rate limit windows

        Raises:
            RateLimitError: If any rate limit is exceeded
        """
        for name, window in self.windows.items():
            if not window.can_make_request():
                retry_after = self._calculate_retry_after(window)
                raise RateLimitError(
                    f"Rate limit exceeded for {self.api_name} ({name})",
                    self.api_name,
                    retry_after
                )

    def record_request(self) -> None:
        """Record a request in all windows"""
        for window in self.windows.values():
            window.add_request()

    def _calculate_retry_after(self, window: RateLimitWindow) -> int:
        """Calculate seconds until the next request can be made"""
        if not window.requests:
            return 0
        oldest_request = window.requests[0]
        return int(oldest_request + window.window_seconds - time.time())

def handle_service_errors(func):
    """Decorator to standardize error handling with RFC 7807"""
    @wraps(func)
    async def wrapper(*args, **kwargs):
        try:
            return await func(*args, **kwargs)
        except ValidationError as e:
            error = create_problem_detail(
                e,
                "Validation Error",
                400,
                func.__name__,
                errors=e.errors()
            )
            logger.error("validation_error", error=error.dict(), exc_info=True)
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail=error.dict(exclude_none=True)
            )
        except DatabaseError as e:
            error = create_problem_detail(
                e,
                "Database Error",
                503,
                func.__name__
            )
            logger.error("database_error", error=error.dict(), exc_info=True)
            raise HTTPException(
                status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
                detail=error.dict(exclude_none=True)
            )
        except APIError as e:
            error = create_problem_detail(
                e,
                "External API Error",
                502,
                func.__name__,
                api=e.api_name
            )
            logger.error("api_error", error=error.dict(), exc_info=True)
            raise HTTPException(
                status_code=status.HTTP_502_BAD_GATEWAY,

            )
        except Exception as e:

    return query.paginate(page=page, per_page=per_page)

def like_recipe(user_id, recipe_id):
    liked = LikedRecipes(user_id=user_id, recipe_id=recipe_id)
    db.session.add(liked)
    db.session.commit()

def unlike_recipe(user_id, recipe_id):
    LikedRecipes.query.filter_by(
        user_id=user_id, recipe_id=recipe_id
    ).delete()
    db.session.commit()
